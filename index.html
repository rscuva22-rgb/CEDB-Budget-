function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Sheet1") || ss.getSheets()[0];
  var data = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Sheet1") || ss.getSheets()[0];
  var payload = JSON.parse(e.postData.contents);
  var contracts = payload.contracts || [];
  
  // Clear existing data rows (keep first 2 header rows)
  var lastRow = sheet.getLastRow();
  if (lastRow > 2) {
    sheet.getRange(3, 1, lastRow - 2, 14).clearContent();
  }
  
  var rows = [];
  contracts.forEach(function(c) {
    var vos = (c.variations && c.variations.length > 0) ? c.variations : [{voNo:'', voNature:'', voAmount:0, voDesc:''}];
    var ipcs = (c.ipcs && c.ipcs.length > 0) ? c.ipcs : [{ipcNo:'', ipcAmount:0, ipcRemarks:''}];
    
    // Determine maximum iterations between IPC and VO rows
    var maxRows = Math.max(vos.length, ipcs.length);
    for (var i = 0; i < maxRows; i++) {
      var vo = vos[i] || {voNo:'', voNature:'', voAmount:0, voDesc:''};
      var ipc = ipcs[i] || {ipcNo:'', ipcAmount:0, ipcRemarks:''};
      
      rows.push([
        c.contractNo || '',
        c.engineer || '',
        c.description || '',
        c.contractSum || 0,
        c.pSum || 0,
        c.contingencies || 0,
        c.advancePayment || 0,
        ipc.ipcNo ? (ipc.ipcNo + ': ' + (ipc.ipcAmount || 0) + (ipc.ipcRemarks ? ' (' + ipc.ipcRemarks + ')' : '')) : (ipc.ipcAmount || 0),
        vo.voNo || '',
        vo.voNature || '',
        vo.voAmount || 0,
        vo.voDesc || '',
        c.issues || ''
      ]);
    }
  });
  
  if (rows.length > 0) {
    sheet.getRange(3, 1, rows.length, 13).setValues(rows);
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
